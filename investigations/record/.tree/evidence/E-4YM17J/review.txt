### Finding 1: label "none" (closest is G-SZCBXZ#9; the rule broken is replay-spec's exit 3 for an unreadable reftable stack)

When git is not on PATH, or cannot open the repository, `refOnDisk` calls the new `reftableHolds`. A malformed table makes it **panic** instead of returning `ok=false`. hyp then exits 2 with a Go stack trace.

The spec line this commit changes says: "a reftable stack hyp cannot read counts as holding it ... every command exits 3". The function's own comment says "the only error is the one that stops a command". AGENTS.md reserves exit 2 for bad arguments.

The cause is in `reftableRef`. Varint values are converted to `int` with no bound check:
- A `prefix` of 1<<63 becomes negative, so `int(prefix) > len(key)` is false and `key[:prefix:prefix]` panics.
- A symref target length near 2^64 makes `p += n + int(l)` negative. The check `p > recEnd` does not catch a negative `p`, and the next `b[p:recEnd]` panics.
- `reftableVarint` can also overflow without any sign.

The fix is to reject any `prefix`, `suffix` or symref length that is negative after conversion or larger than `recEnd - p`, and to reject a negative `p`.

package record

import (
	"fmt"
	"os"
	"path/filepath"
	"testing"
)

// gitVarint encodes v as a reftable varint, as git's put_var_int does.
func gitVarint(v uint64) []byte {
	var buf [10]byte
	i := len(buf) - 1
	buf[i] = byte(v & 0x7f)
	for v >>= 7; v != 0; v >>= 7 {
		v--
		i--
		buf[i] = 0x80 | byte(v&0x7f)
	}
	return append([]byte(nil), buf[i:]...)
}

// table builds a version 1 table whose one ref block holds rec.
func table(rec []byte) []byte {
	b := []byte("REFT\x01\x00\x00\x00")
	b = append(b, make([]byte, 16)...) // min and max update index
	blockLen := 24 + 4 + len(rec) + 3 + 2
	b = append(b, 'r', byte(blockLen>>16), byte(blockLen>>8), byte(blockLen))
	b = append(b, rec...)
	b = append(b, 0, 0, 24) // one restart, at the first record
	b = append(b, 0, 1)     // restart count
	return b
}

// A table hyp cannot read must count as unknown (ok false), so refOnDisk
// counts it as holding the record and the command exits 3. A ref record
// whose varints do not fit an int must not panic.
func TestReftableHoldsDoesNotPanicOnHugeVarints(t *testing.T) {
	neg := uint64(1) << 63
	cases := map[string][]byte{
		"prefix length past int": append(append(gitVarint(neg), gitVarint(1<<3|0)...), append([]byte("a"), gitVarint(0)...)...),
		"symref target length past int": append(append(append(gitVarint(0), gitVarint(1<<3|3)...), 'a'), append(gitVarint(0), gitVarint(^uint64(0)-999)...)...),
	}
	for name, rec := range cases {
		t.Run(name, func(t *testing.T) {
			dir := t.TempDir()
			if err := os.WriteFile(filepath.Join(dir, "tables.list"), []byte("a.ref\n"), 0o644); err != nil {
				t.Fatal(err)
			}
			if err := os.WriteFile(filepath.Join(dir, "a.ref"), table(rec), 0o644); err != nil {
				t.Fatal(err)
			}
			defer func() {
				if r := recover(); r != nil {
					t.Fatalf("reftableHolds panicked: %v", fmt.Sprint(r))
				}
			}()
			holds, ok := reftableHolds(dir, Ref)
			if ok {
				t.Fatalf("a malformed table reads as known: holds %v", holds)
			}
		})
	}
}
