package cli

import (
	"path/filepath"
	"testing"
)

// review5Clones makes a bare remote and two clones a and b of it that share
// one record holding a store s with a Goal, a question and a Hypothesis. It
// returns a, b and the Hypothesis.
func review5Clones(t *testing.T) (a, b, h string) {
	t.Helper()
	gitConfig(t)
	top, err := filepath.EvalSymlinks(t.TempDir())
	if err != nil {
		t.Fatal(err)
	}
	remote := filepath.Join(top, "remote.git")
	gitIn(t, top, "init", "-q", "--bare", remote)
	a = filepath.Join(top, "a")
	gitIn(t, top, "clone", "-q", remote, a)
	writeFile(t, filepath.Join(a, "s", "README.md"), "a store\n")
	gitIn(t, a, "add", "-A")
	gitIn(t, a, "commit", "-qm", "base")
	gitIn(t, a, "push", "-q", "origin", "main")
	t.Chdir(filepath.Join(a, "s"))
	must(t, "init")
	g := must(t, "add", "goal", "--title", "G", "--why", "w")
	q := must(t, "add", "dq", "--parent", g, "--title", "Q", "--why", "w")
	h = must(t, "add", "hyp", "--parent", q, "--title", "H", "--why", "w")
	want(t, hyp(t, "sync"), 0, "synced")
	b = filepath.Join(top, "b")
	gitIn(t, top, "clone", "-q", remote, b)
	t.Chdir(filepath.Join(b, "s"))
	want(t, hyp(t, "sync"), 0, "synced")
	return a, b, h
}

// hfsPath is a path that git on Linux holds by default (core.protectHFS is
// off there) and git on macOS drops from its index (core.protectHFS is on):
// ".git" with a zero-width non-joiner, which HFS+ ignores.
const hfsPart = ".git‌"

// recordInB writes, in clone b with core.protectHFS off as on Linux, a Result
// whose evidence lies below hfsPart. It returns the evidence pointer.
func recordInB(t *testing.T, b, h string) string {
	t.Helper()
	gitIn(t, b, "config", "core.protectHFS", "false")
	t.Chdir(filepath.Join(b, "s"))
	e := must(t, "add", "exp", "--parent", h, "--title", "Eb", "--recipe", "none", "--rubric", "reading", "--why", "w")
	ptr := ".tree/evidence/" + e + "/" + hfsPart + "/out.txt"
	writeFile(t, ptr, "bytes of b\n")
	must(t, "record", "--exp", e, "--verdict", h+"=supports", "--observation", "ran", "--evidence", ptr, "--why", "w")
	want(t, hyp(t, "show", ptr), 0, "bytes of b")
	want(t, hyp(t, "sync"), 0, "synced")
	return ptr
}

// Line 5: a record written where git holds a path (Linux) joins with a clone
// where git drops it (macOS). Each side holds events the other lacks, so
// the join is a merge commit built through the index: hyp sync must still
// give both records the union.
func TestReview5SyncJoinOfAPathOneSidesGitDrops(t *testing.T) {
	a, b, h := review5Clones(t)
	t.Chdir(filepath.Join(a, "s"))
	ea := must(t, "add", "exp", "--parent", h, "--title", "Ea", "--recipe", "none", "--rubric", "reading", "--why", "w")
	ptr := recordInB(t, b, h)
	t.Chdir(filepath.Join(a, "s"))
	r := hyp(t, "sync")
	if r.code != 0 {
		t.Fatalf("hyp sync in a, which holds %s that b lacks, while b's record holds %s: exit %d, stderr %q; want the union", ea, ptr, r.code, r.stderr)
	}
	want(t, hyp(t, "show", ptr), 0, "bytes of b")
}

// Lines 1 and 4: after a fast-forward sync brings that record to the macOS
// clone, a later write there (one that names no such path) must still apply.
func TestReview5WriteAfterFastForwardOfAPathLocalGitDrops(t *testing.T) {
	a, b, h := review5Clones(t)
	ptr := recordInB(t, b, h)
	t.Chdir(filepath.Join(a, "s"))
	want(t, hyp(t, "sync"), 0, "synced")
	want(t, hyp(t, "show", ptr), 0, "bytes of b")
	r := hyp(t, "add", "exp", "--parent", h, "--title", "Ea", "--recipe", "none", "--rubric", "reading", "--why", "w")
	if r.code != 0 {
		t.Fatalf("a write in a after its record took b's %s: exit %d, stderr %q; want exit 0", ptr, r.code, r.stderr)
	}
	want(t, hyp(t, "show", ptr), 0, "bytes of b")
}

// Line 1: a path git can hold, spelled in decomposed Unicode (as a macOS
// tool may spell it), is not refused as a path git cannot hold.
func TestReview5DecomposedPathIsHeld(t *testing.T) {
	recordRepo(t)
	g := must(t, "add", "goal", "--title", "G", "--why", "w")
	q := must(t, "add", "dq", "--parent", g, "--title", "Q", "--why", "w")
	h := must(t, "add", "hyp", "--parent", q, "--title", "H", "--why", "w")
	e := must(t, "add", "exp", "--parent", h, "--title", "E", "--recipe", "none", "--rubric", "reading", "--why", "w")
	ptr := ".tree/evidence/" + e + "/café/out.txt"
	writeFile(t, ptr, "the bytes\n")
	must(t, "record", "--exp", e, "--verdict", h+"=supports", "--observation", "ran", "--evidence", ptr, "--why", "w")
	want(t, hyp(t, "show", ptr), 0, "the bytes")
}
